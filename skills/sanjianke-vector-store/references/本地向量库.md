# 本地向量库：纯 Python 实现、过滤、离线做法

这份文档给出一份**只用 Python 标准库**的向量库实现：一个 JSON 文件 + 余弦相似度 +
元数据过滤。向量化走 `api.a7w.cn` 的网关，其余全部在本地、全部免费。

---

## 1. 向量化：一个 base_url，一把 Key

```python
import json, os, urllib.request

HOST = "https://api.a7w.cn"
KEY = os.environ["A7W_API_KEY"]

# 从 GET /api/v1/models 里挑，别猜；这个模型一旦用于建索引就不能换
EMB_MODEL = "<你挑的 embedding 模型名>"

def embed(texts):
    body = json.dumps({"model": EMB_MODEL, "input": list(texts)}).encode("utf-8")
    req = urllib.request.Request(
        HOST + "/api/v1/embeddings", data=body,
        headers={"Authorization": "Bearer " + KEY,
                 "Content-Type": "application/json"}, method="POST")
    with urllib.request.urlopen(req, timeout=120) as r:
        res = json.loads(r.read().decode("utf-8"))
    if res.get("code") not in (None, 1, 200):      # 成功码是 1
        raise RuntimeError(res.get("msg"))
    return [d["embedding"] for d in res["data"]]
```

**三条纪律**：

1. **索引与查询用同一个模型**。换模型 = 重新向量化全部文本。
2. **算过的向量落盘复用**，不要每次重算。
3. **大批量分批调用**（比如每批 32~64 条），避免单次请求过大。

## 2. 完整实现（可直接复制运行）

```python
# -*- coding: utf-8 -*-
"""零依赖本地向量库：JSON 落盘 + 余弦相似度 + 元数据过滤。"""
import json, math, os

class Store:
    def __init__(self, path="store.json", embed_fn=None):
        self.path = path
        self.embed_fn = embed_fn or embed          # 上一节的 embed 函数
        self.rows = []
        if os.path.exists(path):
            with open(path, encoding="utf-8") as f:
                self.rows = json.load(f)

    # ---------- 落盘 ----------
    def flush(self):
        tmp = self.path + ".tmp"                   # 先写临时文件再替换，防写坏
        with open(tmp, "w", encoding="utf-8") as f:
            json.dump(self.rows, f, ensure_ascii=False)
        os.replace(tmp, self.path)

    # ---------- 读 ----------
    def count(self):
        return len(self.rows)

    def get(self, ids):
        want = set(ids)
        return [r for r in self.rows if r["id"] in want]

    # ---------- 写 ----------
    def add(self, ids, documents, metadatas=None):
        """纯新增语义；id 已存在会覆盖（这里取覆盖，避免重复数据污染检索）。"""
        metadatas = metadatas or [{} for _ in ids]
        vecs = self.embed_fn(documents)
        incoming = {i: (d, m, v) for i, d, m, v in zip(ids, documents, metadatas, vecs)}
        self.rows = [r for r in self.rows if r["id"] not in incoming]
        for i, (d, m, v) in incoming.items():
            self.rows.append({"id": i, "doc": d, "meta": m, "vec": v})
        self.flush()

    def upsert(self, ids, documents, metadatas=None):
        """有则更新、无则新增 —— 与 add 的差别只在语义表达上。"""
        self.add(ids, documents, metadatas)

    def update(self, ids, metadatas):
        """只改元数据，不重算向量（省钱）。"""
        want = dict(zip(ids, metadatas))
        for r in self.rows:
            if r["id"] in want:
                r["meta"] = want[r["id"]]
        self.flush()

    def delete(self, ids):
        want = set(ids)
        before = len(self.rows)
        self.rows = [r for r in self.rows if r["id"] not in want]
        self.flush()
        return before - len(self.rows)

    # ---------- 检索 ----------
    def query(self, text, n=3, where=None):
        qv = self.embed_fn([text])[0]
        rows = [r for r in self.rows if match(r["meta"], where)]
        rows.sort(key=lambda r: cosine(qv, r["vec"]), reverse=True)
        n = max(0, min(n, len(rows)))              # n 比库里条数大也不会报错
        return [{"id": r["id"], "text": r["doc"], "meta": r["meta"],
                 "score": round(cosine(qv, r["vec"]), 4)} for r in rows[:n]]


def cosine(a, b):
    if not a or not b or len(a) != len(b):
        return 0.0
    s = sum(x * y for x, y in zip(a, b))
    na = math.sqrt(sum(x * x for x in a))
    nb = math.sqrt(sum(y * y for y in b))
    return s / (na * nb) if na and nb else 0.0


def match(meta, where):
    """where 语法：{'source': 'kb'} 或 {'source': {'$in': ['kb','wiki']}}，
    以及 {'$and': [...]} / {'$or': [...]}。"""
    if not where:
        return True
    if "$and" in where:
        return all(match(meta, w) for w in where["$and"])
    if "$or" in where:
        return any(match(meta, w) for w in where["$or"])
    for k, cond in where.items():
        v = meta.get(k)
        if isinstance(cond, dict):
            for op, val in cond.items():
                if op == "$eq" and v != val:
                    return False
                if op == "$ne" and v == val:
                    return False
                if op == "$gt" and not (v is not None and v > val):
                    return False
                if op == "$gte" and not (v is not None and v >= val):
                    return False
                if op == "$lt" and not (v is not None and v < val):
                    return False
                if op == "$lte" and not (v is not None and v <= val):
                    return False
                if op == "$in" and v not in val:
                    return False
                if op == "$contains" and (not isinstance(v, str) or val not in v):
                    return False
        elif v != cond:
            return False
    return True
```

### 用法

```python
store = Store("kb.json", embed_fn=embed)

store.add(
    ids=["doc1", "doc2"],
    documents=["退款需在收货后 7 天内发起", "生鲜类不支持无理由退款"],
    metadatas=[{"source": "help-center", "ts": 1700000000},
               {"source": "policy", "ts": 1700000100}],
)

hits = store.query("怎么退款", n=3, where={"source": "help-center"})
hits = store.query("退款", n=3, where={"source": {"$in": ["policy", "help-center"]}})
```

## 3. 元数据怎么用才不踩坑

| 规则 | 说明 |
|---|---|
| **只放简单标量** | 字符串 / 数字 / 布尔。嵌套字典与列表先**拍平成多个字段** |
| 空值要慎用 | 空值过滤行为容易不如预期，宁可别写这个字段 |
| 常用过滤字段**建索引** | 本实现是线性扫描，字段多了会变慢；大库要换服务端 |
| id 用稳定业务键 | 如 `文件名#序号`。用随机串就没法覆盖更新，只会越积越多 |
| 保留来源字段 | `source` / `doc_id` / `ts` 三个字段最常用，建议一开始就加 |

## 4. 离线 / 内网环境的两种做法

| 做法 | 怎么做 | 代价 |
|---|---|---|
| **先算后搬** | 在联网机器上把向量算好落盘，把 `store.json` 拷到离线机器 | 新增内容时要在联网侧补算 |
| **改用关键词检索** | 完全不调用模型，用 BM25 关键词召回 | 没有语义泛化，但零依赖、零成本、完全离线 |

BM25 用纯标准库就能写完，中文按**字 bigram** 切词即可，不需要任何分词库：

```python
import math, re

def tokens(text):
    text = text.lower()
    latin = re.findall(r"[a-z0-9]+", text)
    han = re.findall(r"[\u4e00-\u9fff]", text)
    return latin + (["".join(p) for p in zip(han, han[1:])] or han)

class BM25:
    def __init__(self, docs, k1=1.5, b=0.75):
        self.docs, self.k1, self.b = list(docs), k1, b
        self.tfs = [self._tf(tokens(d)) for d in self.docs]
        self.lens = [sum(tf.values()) for tf in self.tfs]
        self.avg = (sum(self.lens) / len(self.lens)) if self.lens else 0.0
        df = {}
        for tf in self.tfs:
            for w in tf:
                df[w] = df.get(w, 0) + 1
        n = len(self.docs)
        self.idf = {w: math.log(1 + (n - d + 0.5) / (d + 0.5)) for w, d in df.items()}

    @staticmethod
    def _tf(ws):
        tf = {}
        for w in ws:
            tf[w] = tf.get(w, 0) + 1
        return tf

    def search(self, query, top_k=3):
        scored = []
        for i, tf in enumerate(self.tfs):
            s, L = 0.0, (self.lens[i] or 1)
            for w in tokens(query):
                f = tf.get(w)
                if f:
                    s += self.idf.get(w, 0.0) * (f * (self.k1 + 1)) / (
                        f + self.k1 * (1 - self.b + self.b * L / (self.avg or 1)))
            scored.append((s, i))
        scored.sort(reverse=True)
        return [self.docs[i] for _, i in scored[:top_k]]
```

**两路一起用的效果最好**：BM25 负责专有名词与精确匹配，向量负责「换个说法也能搜到」。
落地时把两路结果按分数归一化后合并取 top-k 即可。

## 5. 持久化与并发

| 关注点 | 要点 |
|---|---|
| 原子落盘 | 先写 `.tmp` 再 `os.replace`，避免中途崩掉把文件写坏 |
| 备份 | 定期拷走 JSON；**删除是不可逆的**，没有回收站 |
| 单写者 | 文件型存储是单写者模型。**多个进程各写各的会互相覆盖** |
| 多进程共享 | 统一走服务端形态，让服务端成为唯一写入方 |
| 版本字段 | 在文件里记一个 `schema_version`，以后改结构好做兼容 |

## 6. 什么时候该换更大的库

| 信号 | 说明 |
|---|---|
| 向量规模到**上亿级** | 线性扫描扛不住，需要近邻索引 |
| **高并发**在线检索 | 需要服务端与副本 |
| 需要**细粒度权限、审计、多租户隔离** | 属于托管产品或专门的向量数据库范畴 |
| 想跨表复杂查询 / 事务 / join | 那是关系库的活，别用向量库硬做 |

**迁移成本很低**：写入 / 检索 / 过滤三步的语义不变，把 `Store` 换成服务端客户端，
上层业务代码基本不用改。

## 7. 自检清单

- [ ] embedding 模型已最终确定，**索引与查询用的是同一个**
- [ ] 算过的向量已落盘，没有每次重算
- [ ] 元数据全是简单标量，复杂结构已拍平
- [ ] id 用的是稳定业务键，覆盖更新走 `add`/`upsert` 而不是重复 `add` 新 id
- [ ] `n` 超过库里条数时不会报错（实现里已有 `min` 保护）
- [ ] 落盘是原子的（先写临时文件再替换），并且有定期备份
- [ ] 多进程场景没有让多个实例各写各自的本机文件
- [ ] 离线环境已有「先算后搬」或 BM25 的替代路径
